## 背景

**问题：**
一般的文件读写方式通常是通过 read/write system call 进行对文件的读写。
每一次的读写都需要从用户态变成内核态，对于大文件来说会造成没必要的资源浪费。

**解决方案：**
我们可以将我们需要读写的文件先读取出来，保存到用户的页表中，这样我们之后去访问这个文件就可以直接从用户太进行访问了。
不需要进行频繁的用户态和内核态的装换。

这一次的 LAB 实际上实现的就是 mmap 关于上述读写文件相关的功能，当然 mmap 还有其他功能，本次 LAB 没有涉及。

```c
void *mmap(void *addr, uint64 len, int prot, int flags, int fd, uint64 offset);

int munmap(void *addr, uint64 len);
```

mmap
- addr：用户态地址，代表用户希望映射到的地址，如果为 0 则代表由 mmap 自行决定。
- len： 代表希望映射的文件长度，一般是 PAGESIZE 的整数倍。
- prot： 代表保护规则，比如 PROT_READ PROT_WRITE 来表示映射的 PAGES 的权限
- flags： 代表映射的类型，这里只实现了 MAP_SHARED MAP_PRIVATE 。两者的区别在于 unmmap 的时候是否需要把修改的页（DIRTY PAGE）写回文件。
- fd： 代表要映射的文件的描述符
- offset： 代表要映射的文件的偏移量

munmap
- addr： mmap 返回的地址
- len： 需要 unmap 的长度，可以只 unmap 部分。


基本实现思路：
- 首先在进程的结构体中增加 VMA （virtual memory area）数组，长度固定为 16.代表支持 16 次的 mmap
- 对于每一次的 mmap，其会使用一个 VMA ， 然后遍历当前进程的 PageTable 找到一个合适的用户地址，保存这个地址和其他元数据到 VMA 中（目前还没有分配具体的物理页给这个用户地址）
- 当用户第一次访问这个地址的时候，会出发 PageFault，kernel 回去查询所有的 VMAs，如果找到了符合的 VMA，就根据对应 VMA 中的信息（文件指针， offset）来分配一个 Page 对应指定文件。这之后，用户就像操作内存一样直接操作文件了
- 用户进行 munmap 的时候，同理先找到对应的 VMA，然后进行 uvmunmap 来取消掉对应的 Pagetable 和物理页的映射，并释放物理页。如果 flag 是 MAP_SHARED 就写回文件。如果对应 VNA 的所有 len 区域都被 unmap 了，就释放这个 VMA。

接下来就是具体的 mmap，这里分享一下一些注意点

- vma 需要自己进行 kalloc 分配内存
- 遍历 pagetable 寻找合适的 addr 这一段逻辑比较复杂，需要考虑和其他 vma 里的 addr 有没有重叠
- 需要检查一下文件本身是否 readable 和 writeable，然后和 prot 比较


```c
int va_conflict(struct proc *p, uint64 va, size_t len)
{
  uint64 end;
  if (len > MAXVA - va)
    return -1;
  end = va + len;
  for (int i = 0; i < 16; i++)
  {
    if (!p->vmas[i])
      continue;
    uint64 vstart = (uint64)p->vmas[i]->addr;
    uint64 vend = vstart + p->vmas[i]->len;
    if (!(end <= vstart || va >= vend))
      return 0;
  }
  return -1;
}

void *find_addr(struct proc *p, size_t len)
{
  len = PGROUNDUP(len);
  for (uint64 va = PGSIZE * 6; va + len < MAXVA; va += PGSIZE)
  {
    if (va_conflict(p, va, len) == 0)
      continue; // conflict with existing VMA
    uint64 tmp;
    for (tmp = va; tmp < va + len; tmp += PGSIZE)
    {
      if (walkaddr(p->pagetable, tmp) != 0)
        break;
    }
    if (tmp == va + len)
      return (void *)va;
    va = tmp;
  }
  return 0;
}

void *mmap(void *addr, uint64 len, int prot, int flags, int fd, uint64 offset)
{
  struct proc *p;
  int i;
  p = myproc();
  struct file *f = filedup(p->ofile[fd]);
  
  len = PGROUNDUP(len);
  
  if (flags == MAP_SHARED){
    if (check_prot(prot, p->ofile[fd]) < 0)
      return (void*)-1;
  }

  if (addr == 0)
  {
    addr = find_addr(p, len);
    if (addr == 0) {
      return 0;
    }
  }
  for (i = 0; i < 16; ++i)
  {
    if (p->vmas[i] == 0)
    {
      p->vmas[i] = kalloc();
      p->vmas[i]->addr = addr;
      p->vmas[i]->file_info = f;
      p->vmas[i]->offset = offset;
      p->vmas[i]->flags = flags;
      p->vmas[i]->len = len;
      p->vmas[i]->used = 0;
      p->vmas[i]->prot = prot;
      p->vmas[i]->mapped_pgs = 0;
      p->vmas[i]->bm = kalloc();
      bitmap_init(p->vmas[i]->bm);
      break;
    }
  }
  return addr;
}
```

```c
int check_prot(int prot, struct file* fp){
  if (prot & PROT_READ) {
    if (!fp->readable)
      return -1;
  }
  if (prot & PROT_WRITE){
    if (!fp->writable)
      return -1;
  }
  return 0;
}

```

然后是对应的 pagefault handler：
- 一个 PAGE 只能被 munmap 一次，检查的过程使用 bitmap 进行比较是最方便的办法
- 如果遇到 panic 需要 exit 的时候，检查分配的页是否被释放掉，否则可能会造成内存泄漏
- 注意 mmap 映射文件的偏移量要加上当前地址相对 addr 的偏移量


```c
void pagefault_handler(struct proc *p, uint64 va)
{
  if (va <= 0 || va >= MAXVA)
    exit(-1);
  int vma_idx;
  uint64 vstart;
  int pg_idx;
  va = PGROUNDDOWN(va);
  if ((vma_idx = find_vma_idx(p, va)) < 0)
    exit(-1);
  vstart = (uint64)p->vmas[vma_idx]->addr;
  pg_idx = (va - vstart) / PGSIZE;
  if (bitmap_test(p->vmas[vma_idx]->bm, pg_idx))
    exit(-1);
  bitmap_set(p->vmas[vma_idx]->bm, pg_idx);
  char *mem = kalloc();
  memset((void *)mem, 0, PGSIZE);
  int offset = p->vmas[vma_idx]->offset + (va - vstart);
  file2page(p, vma_idx, mem, va, offset);
}
```

```c
void file2page(struct proc* p, int vma_idx, char* mem, uint64 va, int offset){
  struct inode *ip = p->vmas[vma_idx]->file_info->ip;
  ilock(ip);
  if (readi(ip, 0, (uint64)mem, offset, PGSIZE) < 0){
    kfree((void*)mem);
    panic("file2page");
  }
  iunlock(ip);
  int perm = PTE_U;
  perm |= (p->vmas[vma_idx]->prot << 1);
  if (mappages(p->pagetable, va, PGSIZE, (uint64)mem, perm) < 0)
    kfree((void *)mem);
  p->vmas[vma_idx]->used = 1;
  p->vmas[vma_idx]->mapped_pgs += 1;
  return;
}
```

然后是 munmap，这里和 cow 的实现基本差不多，只不过是变成了写回 file
```c
int munmap(void *addr, uint64 len){
  int i, j;
  struct proc *p;
  p = myproc();
  uint64 start, end, vstart, vend;
  start = (uint64)addr;
  end = start + len;
  int npages;
  for (i = 0; i < 16; ++i)
  {
    if (p->vmas[i] != 0){
      vstart = (uint64)p->vmas[i]->addr;
      vend = vstart + p->vmas[i]->len;
      if (!(start >= vstart && end <= vend))
        continue;
      if (p->vmas[i]->flags == MAP_SHARED){
        filewrite_offset(p->vmas[i]->file_info, start, len, p->vmas[i]->offset + (start - vstart));
      }
      npages = len / PGSIZE;
      if (p->vmas[i]->mapped_pgs > 0) {
        for (j = 0; j < npages; ++j){
          if (walkaddr(p->pagetable, start))
          {
            uvmunmap(p->pagetable, start, 1, 1);
            p->vmas[i]->mapped_pgs--;
          }
          start += PGSIZE;
        }
      }

      if (p->vmas[i]->mapped_pgs <= 0)
      {
        fileclose(p->vmas[i]->file_info);
        p->vmas[i]->used = 0;
        kfree((void *)p->vmas[i]->bm);
        kfree((void *)p->vmas[i]);
        p->vmas[i] = (void *)0;
      }
      return 0;
    }
  }
  return 0;
}
```
